// Laravel 12 - Merchant Dashboard & KYC Controller
namespace App\Http\Controllers;

use App\Services\KycVerificationService;
use App\Models\Merchant;
use Illuminate\Http\JsonResponse;

class MerchantDashboardController extends Controller {
    public function verifyMerchantKyc(Request $request): JsonResponse {
        $merchant = Merchant::findOrFail($request->merchant_id);
        
        // 1. Validate PAN, Aadhaar, GST documents
        $kycResult = KycVerificationService::verify([
            'pan_number'   => $request->pan,
            'aadhaar_id'   => $request->aadhaar,
            'gst_number'   => $request->gst,
            'business_doc' => $request->file('document')
        ]);
        
        // 2. Update merchant verification status
        $merchant->update([
            'kyc_status' => $kycResult->isApproved() ? 'VERIFIED' : 'PENDING_REVIEW',
            'kyc_score'  => $kycResult->score,
            'verified_at'=> now()
        ]);
        
        return response()->json([
            'status' => 'SUCCESS',
            'merchant_id' => $merchant->id,
            'kyc_status'  => $merchant->kyc_status
        ], 200);
    }
}
